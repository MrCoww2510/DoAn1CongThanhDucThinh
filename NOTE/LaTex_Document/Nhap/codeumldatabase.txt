entity "NguoiDung" as ND {
    * MaND : VARCHAR(20) <<PK>>
    --
    * HoTen : VARCHAR(100)
    SDT : VARCHAR(15) <<NULL>>
    Email : VARCHAR(100) <<NULL>>
    DiaChiNhanHang : VARCHAR(255) <<NULL>>
}

entity "TaiKhoan" as TK {
    * MaTK : VARCHAR(20) <<PK>>
    --
    * MaND : VARCHAR(20) <<FK, UNIQUE>>
    * TenDangNhap : VARCHAR(50) <<UNIQUE>>
    * MatKhau : VARCHAR(255)
    * VaiTro : VARCHAR(20)
    * TrangThai : VARCHAR(30) <<DEFAULT 'HoatDong'>>
}

entity "DanhMuc" as DM {
    * MaDM : VARCHAR(20) <<PK>>
    --
    * TenDM : VARCHAR(100) <<UNIQUE>>
}

entity "SanPham" as SP {
    * MaSP : VARCHAR(20) <<PK>>
    --
    * MaDM : VARCHAR(20) <<FK>>
    * TenSP : VARCHAR(200)
    * GiaBan : DECIMAL(18,2)
    * SoLuongTon : INT <<DEFAULT 0>>
}

entity "ChiTietSanPham" as CTSP {
    * MaCTSP : VARCHAR(20) <<PK>>
    --
    * MaSP : VARCHAR(20) <<FK, UNIQUE>>
    LoaiSP : VARCHAR(100) <<NULL>>
    ThongSoKT : TEXT <<NULL>>
    ThuongHieu : VARCHAR(100) <<NULL>>
    HinhAnh : VARCHAR(500) <<NULL>>
    MoTa : TEXT <<NULL>>
    TrangThai : VARCHAR(50) <<NULL>>
}

entity "GioHang" as GH {
    * MaGH : VARCHAR(20) <<PK>>
    --
    * MaND : VARCHAR(20) <<FK, UNIQUE>>
    * TongTien : DECIMAL(18,2) <<DEFAULT 0>>
}

entity "ChiTietGioHang" as CTGH {
    * MaCTGH : VARCHAR(20) <<PK>>
    --
    * MaGH : VARCHAR(20) <<FK>>
    * MaSP : VARCHAR(20) <<FK>>
    * SoLuong : INT
    ..
    UNIQUE UQ_GioHang_SanPham (MaGH, MaSP)
}

entity "DonHang" as DH {
    * MaDH : VARCHAR(20) <<PK>>
    --
    * MaND : VARCHAR(20) <<FK>>
    MaNV : VARCHAR(20) <<FK, NULL>>
    * TenNguoiNhan : VARCHAR(100)
    * DiaChiGiao : VARCHAR(255)
    * SDTNhan : VARCHAR(15)
    * TongTien : DECIMAL(18,2)
    * PhuongThucTT : VARCHAR(20)
    * TrangThaiDH : VARCHAR(50)
    * TrangThaiTT : VARCHAR(50)
    * NgayTao : DATETIME <<DEFAULT CURRENT_TIMESTAMP>>
}

entity "ChiTietDonHang" as CTDH {
    * MaCTDH : VARCHAR(20) <<PK>>
    --
    * MaDH : VARCHAR(20) <<FK>>
    * MaSP : VARCHAR(20) <<FK>>
    * SoLuong : INT
    * DonGia : DECIMAL(18,2)
}

entity "GiaoDichThanhToan" as GDTT {
    * MaGD : VARCHAR(50) <<PK>>
    --
    * MaDH : VARCHAR(20) <<FK>>
    * SoTien : DECIMAL(18,2)
    * TrangThai : VARCHAR(30)
    ThoiGianGiaoDich : DATETIME <<NULL>>
}

entity "LichSuKho" as LSK {
    * MaLS : VARCHAR(20) <<PK>>
    --
    * MaSP : VARCHAR(20) <<FK>>
    * SoLuongThayDoi : INT
    * LoaiThaoTac : VARCHAR(30)
    * MaND : VARCHAR(20) <<FK>>
    * ThoiGian : DATETIME <<DEFAULT CURRENT_TIMESTAMP>>
}

note as Rules
  Dấu * biểu thị NOT NULL; cột không có dấu * cho phép NULL.
  Mỗi bảng có ENGINE=InnoDB, DEFAULT CHARSET=utf8mb4.
  FK không ghi ON DELETE/ON UPDATE dùng hành vi mặc định MySQL;
  chỉ hai FK hiển thị CASCADE có ON DELETE CASCADE.
  SQL không khai báo ON UPDATE, AUTO_INCREMENT hay index riêng.
  CHECK trạng thái đơn hàng được giữ theo biểu thức HEX trong SQL;
  sơ đồ ghi số lượng giá trị, không thay bằng quy tắc suy đoán.
end note
